async function callEntities(entities) {
  const body = JSON.stringify({ FUNC: "GET_DATA", entities, firstTime: 0, session: SESSION });
  const res = await fetch(BASE_URL, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      'Origin': 'https://rastreo.idecgps.com.ar',
      'Referer': 'https://rastreo.idecgps.com.ar/streetz/',
      'User-Agent': 'Mozilla/5.0',
      'Cookie': COOKIE
    },
    body
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}

function decode(raw){
  const fields = raw.URIEncodedFields || [];
  const dataObj = raw.data || {};
  let positions = dataObj.lastPositions || dataObj.positions || dataObj.vehicles || [];
  const decoded = positions.map(row => {
    if (!Array.isArray(row)) return row;
    const obj = {};
    fields.forEach((f,i)=> obj[f]=row[i]);
    return obj;
  });
  return { fields, positions: decoded, rawData: dataObj };
}

app.get('/api/vehicles', async (req, res) => {
  try {
    // Pide posiciones vivas
    let raw = await callEntities({
      lastPositions: true, positions: false, vehiclesTypes: false, vehiclesStates: false,
      companies: false, gps: false, pointsOfInterest: false, pointsOfInterestCategories: false,
      geofences: false, geofencesCategories: false, eventsTypes: false, routes: false,
      routesCategories: false, routePoints: false, vehiclesEvents: false, commercialRepresentatives: false
    });
    let { fields, positions } = decode(raw);

    if (positions.length === 0) {
      raw = await callEntities({ lastPositions: false, positions: true, vehiclesTypes: false, vehiclesStates: false, companies:false,gps:false,pointsOfInterest:false,pointsOfInterestCategories:false,geofences:false,geofencesCategories:false,eventsTypes:false,routes:false,routesCategories:false,routePoints:false,vehiclesEvents:false,commercialRepresentatives:false });
      ({ fields, positions } = decode(raw));
    }

    let vehicles = positions.map(v => ({
      id: v.vehi_id || v.gps_id || v.id,
      patente: v.vehi_patente || v.patente || '',
      alias: v.vehi_alias || v.label || '',
      chofer: v.poul_chofer || v.chofer || '',
      lat: parseFloat(v.vehi_latitud || v.latitud || v.lat || v.y),
      lng: parseFloat(v.vehi_longitud || v.longitud || v.lng || v.x),
      velocidad: v.velocidad || v.vehi_velocidad || 0,
      estado: v.estado_text || v.estado || '',
      evento: v.evento_text || v.evento || '',
      ultima: v.fecha || v.ultima || ''
    })).filter(v => !isNaN(v.lat) && !isNaN(v.lng));

    // --- FILTRO 38-46 ---
    // Filtra por alias o patente que contenga 38,39,40,41,42,43,44,45,46
    // Si tu flota usa alias tipo "Interno 38" o "Camion 38", lo agarra. Si no, cambia acá por patente exacta.
    const filtrados = vehicles.filter(v => {
      const texto = `${v.alias} ${v.patente}`.toLowerCase();
      return SOLO.some(n => texto.includes(n) || texto.includes(` ${n}`) || v.alias == n || v.patente == n);
    });

    // Si el filtro por nombre no encontró nada (porque usan IDs internos distintos), 
    // por defecto te devuelve los últimos 9 por ID para que veas el mapa funcionando.
    const finalList = filtrados.length > 0 ? filtrados : vehicles.slice(-9);

    res.json({ count_total: vehicles.length, count_filtrado: finalList.length, solo_buscados: SOLO, fields, vehicles: finalList });
  } catch (e) {
    res.status(500).json({ error: e.message });
  }
});
